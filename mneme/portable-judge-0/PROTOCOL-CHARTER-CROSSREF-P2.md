1. **Nothing is fixed.** Not the protocol, not the charter, not one byte. Rewrapping frozen

**Honest limit of this check.** It verifies the **five sentences named above**, no others. It

convention*, disclosed here so anyone may object to it; the raw hashes are given so the
objection can be made against the same bytes.

import re, hashlib
P = 'portable-judge-0/PROTOCOL-CANDIDATE-P1.md'
C = 'LANGUAGEHOOD-AND-SUCCESSION-CHARTER-0-CANDIDATE-P1.md'

def blocks(path):                      # every maximal run of '> ' lines
    out, cur = [], []
    for l in open(path, encoding='utf-8').read().split('\n'):
        if l.startswith('> '):
            cur.append(l)
        else:
            if cur: out.append('\n'.join(cur)); cur = []
    if cur: out.append('\n'.join(cur))
    return out

def norm(b):                           # strip '> ', join, collapse whitespace
    return re.sub(r'\s+', ' ', ' '.join(x[2:].strip() for x in b.split('\n'))).strip()

for key in ['Given only the frozen public normative packet',
            'A clean-room non-Common-Lisp implementation of the Many Acts /0 language layer,']:
    p = [b for b in blocks(P) if key in norm(b)][0]

    print('raw  equal:', p.encode() == c.encode(),
          '| norm equal:', norm(p) == norm(c),
          '| norm sha256:', hashlib.sha256(norm(p).encode()).hexdigest())
